# ============================================================
# Encabezados de seguridad HTTP
# ============================================================
<IfModule mod_headers.c>
  # Evita clickjacking
  Header always set X-Frame-Options "SAMEORIGIN"

  # Protege contra XSS adivinando tipos de archivo
  Header always set X-Content-Type-Options "nosniff"

  # Referrer policy estricta
  Header always set Referrer-Policy "strict-origin-when-cross-origin"

  # Fuerza HTTPS
  Header always set Strict-Transport-Security "max-age=63072000; includeSubDomains; preload"

  # Opener policy
  Header always set Cross-Origin-Opener-Policy "same-origin"

  # Desactiva permisos no usados
  Header always set Permissions-Policy "geolocation=(), microphone=(), camera=()"
  # Política de Seguridad de Contenido (CSP)
  # Incluye Google Maps/Places, Google Tag Manager/Analytics y Supabase.
  # Sin hash fijo de script: Next.js genera un script inline distinto en
  # cada build (datos de hidratación), así que un hash fijo lo bloquea
  # siempre. Con 'unsafe-inline' sin hash, el navegador sí lo permite.
  Header always set Content-Security-Policy "default-src 'self'; script-src 'self' 'unsafe-inline' https://www.googletagmanager.com https://maps.googleapis.com https://maps.gstatic.com; style-src 'self' 'unsafe-inline' https://fonts.googleapis.com; font-src 'self' https://fonts.gstatic.com; img-src 'self' data: https:; connect-src 'self' https://*.supabase.co https://*.supabase.in https://maps.googleapis.com https://www.google-analytics.com https://www.googletagmanager.com; frame-ancestors 'self'; object-src 'none'; base-uri 'self'"
</IfModule>
